task_id: "TASK-2026-VU-DEPLOY"
node: PROD  # 生产上线与运维
updated: 2026-08-09

exception_events:

  - event_id: PROD-001
    node_failed: 产物下载
    error_class: ENCODING        # Python http.server 编码限制
    status: RESOLVED
    symptom: "点击下载产物按钮，前端显示'失败'，后端无 HTTP 响应（连接中断）"
    root_cause:
      statement: "py http.server.send_header() 仅支持 latin-1 编码，产物文件名含中文字符（如 1483589377_去重.mp4）导致 UnicodeEncodeError，Handler 直接崩溃"
      crash_trace: "File .../http/server.py, line 522, in send_header: ('%s: %s\r\n' % (keyword, value)).encode('latin-1', 'strict') -> UnicodeEncodeError"
      repro: "curl 'http://localhost:8765/local/download/1483589377_%E5%8E%BB%E9%87%8D.mp4' -> HTTP 000 (连接中断)"
      contrast: "英文文件名 test123.mp4 -> HTTP 200 正常"
    resolution:
      action_taken: "Content-Disposition 改用 RFC 5987 编码"
      before: 'f\'attachment; filename="{safe.name}"\''
      after: "f\"attachment; filename*=UTF-8''{quote(safe.name)}\""
      commit: 23c34c9
    lesson: "Python http.server 的 HTTP 头管线是 latin-1，任何非 ASCII 值必须 URL-encode 后放入 header。不限于文件名——所有自定义头值都要验证编码安全。"

  - event_id: PROD-002
    node_failed: HTTPS 部署后前端无法连接 MCP
    error_class: CONFIG         # 前端硬编码 http:// 与 HTTPS 不兼容
    status: RESOLVED
    symptom: "添加 Let's Encrypt 证书 + nginx 443 反代后，浏览器访问 https://vu.evenblue.top 显示'无法连接 MCP Server'"
    root_cause:
      statement: "app.js 中 API_BASE 硬编码 `http://` 协议前缀。HTTPS 页面加载后 JS 计算 API_BASE = http://vu.evenblue.top，浏览器混合内容安全策略拦截 fetch() 请求（https 页面禁止加载 http 资源）"
      code_defect: "return `http://${h}`  // 不管实际页面协议，永远返回 http"
      impact: "fetch 请求被浏览器拦截，根本未到达 nginx，前端误以为后端未启动"
    resolution:
      action_taken: "API_BASE 改用 window.location.protocol 动态取值"
      before: 'return `http://${h}`'
      after: 'const proto = window.location.protocol; return `${proto}//${h}`'
      commit: edb8451
    lesson: "前端 API_BASE 绝不能硬编码协议。部署 HTTP 时正常工作 ≠ HTTPS 也能工作。混合内容拦截是浏览器级别静默失败，不会在 nginx access log 留痕，只能通过浏览器 DevTools Console 发现。"

  - event_id: PROD-003
    node_failed: 服务器仓库与线上不一致
    error_class: PROCESS        # 部署流程缺失版本追踪
    status: RESOLVED
    symptom: "GitHub main 分支缺少线上已运行的功能（上传/下载/临时目录），仓库 ≠ 服务器"
    root_cause:
      statement: "服务器通过 SFTP 上传文件部署，非 git checkout；部署后未将改动同步回 main 分支，导致仓库滞后于线上"
      gap: "main (c303ef5) 未含 dev 已提交的 4 个 feature commit (36a55f7/cb0f8dd/275584d/2dc55e0)，而服务器运行的是 SFTP 快照等价于 dev 的功能态"
    resolution:
      action_taken: "dev fast-forward merge 进 main，推至 origin/main"
      after: "main = dev = 服务器功能态"
    lesson: "SFTP 部署的项目必须显式维护'哪个分支 = 线上版本'的对应关系。部署后立即把对应分支推至仓库，避免仓库与线上脱节。"

historical_lessons_deploy:
